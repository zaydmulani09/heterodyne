def fibonacci(n):
    """Return the n-th Fibonacci number."""
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

def is_prime(n):
    if n < 2:
        return False
    for d in range(2, int(n ** 0.5) + 1):
        if n % d == 0:
            return False
    return True

Question: A water tank holds 1200 liters. A pipe fills it at 40 liters per minute while a leak drains 15 liters per minute. Starting empty, how long until it is full?
Answer: The tank gains 40 - 15 = 25 liters each minute. 1200 / 25 = 48. It is full after 48 minutes.

    def __init__(self):
        self.items = []

    def push(self, item):
        self.items.append(item)
